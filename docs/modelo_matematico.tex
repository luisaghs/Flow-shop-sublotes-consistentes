\section{El problema}

La planta recibe órdenes de trabajo. Cada orden pasa, en este orden, por
cuatro etapas: \textbf{troquelado, ensamble, costura y volteado}. En cada etapa
hay varias máquinas que trabajan en paralelo, y no todas tienen la misma
velocidad.

Para no esperar a que toda la orden termine una etapa, cada orden se divide en
\textbf{sublotes} (partes más pequeñas). Cada sublote pasa a la siguiente etapa
apenas termina la actual. Un sublote tiene \textbf{el mismo tamaño en todas las
etapas}; por eso se llaman \emph{sublotes consistentes}.

\textbf{Objetivo:} que las órdenes se atrasen lo menos posible (menor tardanza
total). Si dos programaciones tienen la misma tardanza, se prefiere la que
termina antes todo el trabajo (menor \emph{makespan}).

\subsection*{Supuestos}
\begin{enumerate}
  \item El tiempo se mide en horas hábiles (sin noches, domingos ni festivos).
        Todas las máquinas están libres al inicio.
  \item Una máquina hace un solo sublote a la vez y no se interrumpe.
  \item Un sublote se hace en una sola máquina por etapa. Dos sublotes de la
        misma orden sí pueden estar en máquinas distintas al mismo tiempo.
  \item Un sublote puede entrar a la siguiente etapa apenas sale de la
        anterior (no hay tiempo de transporte ni límite de espacio entre etapas).
  \item Cualquier máquina de una etapa puede hacer cualquier referencia
        (así está en la base de datos), aunque con tiempos distintos.
  \item En troquelado hay un \textbf{alistamiento} de 4,98 min cuando la
        máquina cambia de referencia y antes de su primer sublote. Se puede
        hacer mientras la máquina espera que llegue el sublote.
  \item El tiempo de un sublote es proporcional a su tamaño.
\end{enumerate}

\section{Notación}

\subsection*{Conjuntos}

j \in \J & Órdenes de trabajo. \\

m \in \M_s & Máquinas de la etapa $s$. \\
b = 1,\dots,B & Sublotes de cada orden ($B = 10$). \\
i \in \N & Todos los sublotes. Cada sublote es una pareja $i=(j,b)$: el sublote $b$ de la orden $j$. \\

\subsection*{Datos}
\begin{center}
\begin{tabular}{>{$}l<{$} p{12.5cm}}
\toprule
Q_j & Cantidad de la orden $j$ [docenas]. \\
r_j & Hora en que la orden $j$ puede empezar (liberación) [h]. \\
d_j & Hora de entrega de la orden $j$ [h]. \\
a_{jm} & Horas que tarda la máquina $m$ en hacer una docena de la orden $j$
         ($=$ min por docena $\div$ 60). \\
\sigma_{ii'} & Alistamiento en troquelado si el sublote $i'$ va justo después de $i$ en
               la misma máquina: $4{,}98/60$ h si son de referencias distintas y $0$ si
               son de la misma referencia. \\
\sigma_{0} & Alistamiento antes del primer sublote de una máquina de troquelado ($4{,}98/60$ h). \\
H & Un número muy grande (por ejemplo, la suma de todos los tiempos). \\
\bottomrule
\end{tabular}
\end{center}
En las etapas 2, 3 y 4 no hay alistamiento ($\sigma = 0$).

\subsection*{Variables (lo que el modelo decide)}
\begin{center}
\begin{tabular}{>{$}l<{$} p{12.5cm}}
\toprule
x_{jb} & Tamaño del sublote $b$ de la orden $j$ [docenas, número entero]. \\
y_{ism} & Vale 1 si el sublote $i$ se hace en la máquina $m$ de la etapa $s$; si no, 0. \\
z_{ii'sm} & Vale 1 si en la máquina $m$ de la etapa $s$ el sublote $i'$ va
            \emph{justo después} de $i$; si no, 0. Si $i=0$, quiere decir que
            $i'$ es el primero de la máquina. \\
w_{ism} & Docenas del sublote $i$ que hace la máquina $m$ en la etapa $s$
          (vale $x_{jb}$ si se usa esa máquina y 0 si no). \\
S_{is},\ C_{is} & Hora de inicio y hora de fin del sublote $i$ en la etapa $s$. \\
C_j & Hora en que termina la orden $j$ (cuando sale su último sublote de volteado). \\
T_j & Tardanza de la orden $j$: horas de atraso (0 si llega a tiempo). \\
C_{\max} & Makespan: hora en que termina la última orden. \\
\bottomrule
\end{tabular}
\end{center}

\section{Función objetivo}
\begin{equation}
  \text{Minimizar}\quad \sum_{j\in\J} T_j
  \qquad\text{y, si hay empate,}\qquad \text{minimizar}\quad C_{\max}
  \label{eq:obj}
\end{equation}
\enpalabras{primero, que la suma de los atrasos sea lo más pequeña posible;
después, terminar todo lo antes posible.}

\section{Restricciones}

\paragraph{1. Tamaño de los sublotes.}
\begin{align}
  \sum_{b=1}^{B} x_{jb} &= Q_j && \text{para cada orden } j \label{eq:suma}\\
  x_{jb} &\ge 1 && \text{para cada sublote} \label{eq:min}
\end{align}
\enpalabras{los sublotes de una orden suman la cantidad pedida, y cada sublote
tiene al menos una docena. Como $x_{jb}$ no depende de la etapa, el sublote
tiene el mismo tamaño en todas las etapas.}

\paragraph{2. Cada sublote va a una máquina por etapa.}
\begin{align}
  \sum_{m\in\M_s} y_{ism} &= 1 && \text{para cada sublote } i \text{ y etapa } s \label{eq:asig}
\end{align}

\paragraph{3. Cantidad que hace cada máquina.}
\begin{align}
  w_{ism} &\le Q_j\, y_{ism} \label{eq:w1}\\
  w_{ism} &\le x_{jb} \label{eq:w2}\\
  w_{ism} &\ge x_{jb} - Q_j\,(1-y_{ism}) && \text{para cada sublote } i=(j,b),\ \text{etapa } s,\ \text{máquina } m \label{eq:w3}
\end{align}
\enpalabras{si el sublote va a la máquina $m$ ($y=1$), entonces $w = x$; si no
va ($y=0$), entonces $w = 0$. Se hace así porque multiplicar dos variables
($x \cdot y$) no está permitido en un modelo lineal.}

\paragraph{4. Hora de fin de cada sublote.}
\begin{align}
  C_{is} &= S_{is} + \sum_{m\in\M_s} a_{jm}\, w_{ism} && \text{para cada sublote } i \text{ y etapa } s \label{eq:fin}
\end{align}
\enpalabras{fin = inicio + tiempo de proceso en la máquina elegida.}

\paragraph{5. Orden de las etapas.}
\begin{align}
  S_{i1} &\ge r_j && \text{para cada sublote } i=(j,b) \label{eq:rel}\\
  S_{i,s+1} &\ge C_{is} && \text{para cada sublote } i \text{ y etapa } s < |\Sg| \label{eq:prec}
\end{align}
\enpalabras{un sublote no empieza troquelado antes de que se libere su orden,
y no empieza una etapa hasta terminar la anterior. Cada sublote avanza por su
cuenta, sin esperar a los demás sublotes de su orden.}

\paragraph{6. Secuencia en cada máquina.}
\begin{align}
  \sum_{i \in \{0\}\cup\N,\ i\neq i'} z_{ii'sm} &= y_{i'sm} && \text{para cada } s,\ m,\ i' \label{eq:ant}\\
  \sum_{i' \in \N,\ i'\neq i} z_{ii'sm} &\le y_{ism} && \text{para cada } s,\ m,\ i \label{eq:suc}\\
  \sum_{i' \in \N} z_{0i'sm} &\le 1 && \text{para cada } s,\ m \label{eq:primero}
\end{align}
\enpalabras{en cada máquina los sublotes forman una fila: cada sublote tiene
uno justo antes (otro sublote o el inicio de la máquina) y como máximo uno
justo después, y cada máquina tiene como máximo un primer sublote.}

\paragraph{7. Sin choques en las máquinas y alistamientos.}
\begin{align}
  S_{i's} &\ge C_{is} + \sigma_{ii'} - H\,(1 - z_{ii'sm}) && \text{para cada } s,\ m,\ i \neq i' \label{eq:disy}\\
  S_{i'1} &\ge \sigma_{0}\, z_{0i'1m} && \text{para cada } m \in \M_1,\ i' \label{eq:setup0}
\end{align}
\enpalabras{si $i'$ va justo después de $i$ en una máquina, $i'$ empieza
cuando termina $i$ más el alistamiento (si cambia la referencia). Si $i'$ no va
después de $i$, el término con $H$ desactiva la restricción. El primer sublote
de cada máquina de troquelado espera su alistamiento inicial. Como el
alistamiento se cuenta desde que la máquina queda libre, se puede hacer
mientras el sublote todavía no ha llegado.}

\paragraph{8. Fin de la orden, atraso y makespan.}
\begin{align}
  C_j &\ge C_{(j,b),|\Sg|} && \text{para cada sublote } b \text{ de la orden } j \label{eq:cj}\\
  T_j &\ge C_j - d_j,\qquad T_j \ge 0 && \text{para cada orden } j \label{eq:tj}\\
  C_{\max} &\ge C_j && \text{para cada orden } j \label{eq:cmax}
\end{align}
\enpalabras{la orden termina cuando sale de volteado su último sublote; el
atraso es lo que pasa de la fecha de entrega (o cero si llega a tiempo); el
makespan es el fin de la última orden.}

\paragraph{9. Tipo de variables.}
\[
  x_{jb} \text{ enteros};\quad y_{ism},\ z_{ii'sm} \in \{0,1\};\quad
  w_{ism},\ S_{is},\ C_{is},\ C_j,\ T_j,\ C_{\max} \ge 0 .
\]

\paragraph{Horas y fechas.} Las horas se cuentan sobre el calendario laboral.
Para pasar una hora a fecha se busca el primer día en que las horas hábiles
acumuladas alcanzan ese valor.

\section{Cómo se resuelve: algoritmo genético}

El modelo exacto es muy grande para esta planta (100 sublotes y 36 máquinas
dan cerca de 360\,000 variables $z$), así que se usa un \textbf{algoritmo
genético}. Este busca buenas soluciones probando muchas programaciones y
combinando las mejores. Todas las soluciones que construye cumplen las
restricciones de arriba, aunque no se garantiza que sean la mejor posible.

\paragraph{Cómo se representa una solución.} Cada solución tiene dos partes:
\begin{enumerate}
  \item \textbf{Tamaños:} para cada orden, qué porcentaje de la cantidad va en
        cada sublote. Por ejemplo, 12\,\%, 8\,\%, 10\,\%\dots{} (suman 100\,\%).
        Esos porcentajes se pasan a docenas enteras de modo que cada sublote
        tenga al menos una docena y la suma sea exacta.
  \item \textbf{Prioridades:} para cada etapa, una lista con el orden en que
        se programan los sublotes.
\end{enumerate}

\paragraph{Cómo se arma la programación.} Etapa por etapa, y dentro de cada
etapa en el orden de la lista:
\begin{enumerate}
  \item El sublote está listo cuando se libera su orden (troquelado) o
        cuando termina la etapa anterior.
  \item Se prueba en cada máquina de la etapa:
    \begin{itemize}
      \item En troquelado se pone después del último sublote de la máquina,
            sumando el alistamiento si cambia la referencia.
      \item En las otras etapas se pone en el primer espacio libre en que quepa.
    \end{itemize}
  \item Se elige la máquina donde \textbf{termina más temprano}.
\end{enumerate}
Al final se calcula la tardanza total y el makespan de esa programación.

\paragraph{Cómo evoluciona.}
\begin{itemize}
  \item \textbf{Población inicial:} 150 soluciones al azar. Una de ellas se
        cambia por la mejor regla de despacho (FIFO, EDD, SPT, LPT, MST o CR)
        con sublotes iguales.
  \item \textbf{Selección:} se toman 3 soluciones al azar y gana la mejor. La
        mejor de cada generación pasa siempre a la siguiente.
  \item \textbf{Cruce} (90\,\% de las veces), que mezcla dos soluciones padre:
    \begin{itemize}
      \item Tamaños: se toma un número $\lambda$ al azar entre 0 y 1. Un hijo
            recibe $\lambda$ del padre 1 y $1-\lambda$ del padre 2; el otro hijo, al revés.
      \item Prioridades: el hijo copia un tramo de la lista de un padre y
            completa el resto en el orden en que aparecen en la lista del otro padre.
    \end{itemize}
  \item \textbf{Mutación} (50\,\% de los hijos), que hace cambios pequeños:
    \begin{itemize}
      \item Con 30\,\% de probabilidad, intercambia dos sublotes en la lista de una etapa.
      \item En cada orden, con 40\,\% de probabilidad, sube o baja un poco el
            porcentaje de un sublote.
    \end{itemize}
  \item \textbf{Parada:} cuando pasan 50 generaciones sin mejorar, o al llegar a 1000.
\end{itemize}
